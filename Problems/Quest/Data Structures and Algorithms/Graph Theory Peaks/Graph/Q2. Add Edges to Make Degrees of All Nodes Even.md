# Intuition

For all nodes to have even degree, we need to fix nodes with odd degree. The key insight is that **the number of odd-degree nodes must be even** (a fundamental graph theory property - the handshaking lemma).

**Key observations:**
1. Adding an edge increases the degree of exactly 2 nodes by 1
2. We can add at most 2 edges (affecting at most 4 nodes)
3. Nodes with odd degree need to be paired up
4. We need to check if valid pairings exist without creating duplicate edges

---

# Approach

1. **Count odd-degree nodes:**
   - Calculate degree of each node
   - Collect nodes with odd degree

2. **Handle cases based on count of odd-degree nodes:**
   - **0 odd nodes:** Already valid, return true
   - **2 odd nodes:** Add 1 edge between them (if not duplicate)
   - **4 odd nodes:** Add 2 edges to pair them up
   - **Other counts:** Impossible (must be even, and ≤4 with 2 edges)

3. **Validation for each case:**
   - Check if edges don't already exist
   - For 2 odd nodes: try direct connection or via intermediate node
   - For 4 odd nodes: try all valid pairings

---

# Complexity

- **Time complexity:** $O(n + e)$ where e is number of edges
- **Space complexity:** $O(n + e)$ for adjacency set and degree array

```typescript
function isPossible(n: number, edges: number[][]): boolean {
    // Step 1: Calculate degree of each node
    const degree = new Array(n + 1).fill(0);
    const edgeSet = new Set<string>();
    
    for (const [a, b] of edges) {
        degree[a]++;
        degree[b]++;
        // Store edges in both directions for quick lookup
        edgeSet.add(`${Math.min(a, b)},${Math.max(a, b)}`);
    }
    
    // Step 2: Find nodes with odd degree
    const oddNodes: number[] = [];
    for (let i = 1; i <= n; i++) {
        if (degree[i] % 2 === 1) {
            oddNodes.push(i);
        }
    }
    
    // Step 3: Handle different cases
    const oddCount = oddNodes.length;
    
    // Case 1: Already all even degrees
    if (oddCount === 0) {
        return true;
    }
    
    // Case 2: Exactly 2 odd-degree nodes
    if (oddCount === 2) {
        return canConnect2Nodes(oddNodes, edgeSet, n);
    }
    
    // Case 3: Exactly 4 odd-degree nodes
    if (oddCount === 4) {
        return canConnect4Nodes(oddNodes, edgeSet);
    }
    
    // Case 4: Any other count (impossible)
    return false;
}

function hasEdge(a: number, b: number, edgeSet: Set<string>): boolean {
    return edgeSet.has(`${Math.min(a, b)},${Math.max(a, b)}`);
}

function canConnect2Nodes(oddNodes: number[], edgeSet: Set<string>, n: number): boolean {
    const [a, b] = oddNodes;
    
    // Option 1: Direct edge between a and b
    if (!hasEdge(a, b, edgeSet)) {
        return true;
    }
    
    // Option 2: Connect via an intermediate node with even degree
    // Add edges a-x and b-x (x must have even degree and not connected to a or b)
    for (let x = 1; x <= n; x++) {
        if (x !== a && x !== b) {
            if (!hasEdge(a, x, edgeSet) && !hasEdge(b, x, edgeSet)) {
                return true;
            }
        }
    }
    
    return false;
}

function canConnect4Nodes(oddNodes: number[], edgeSet: Set<string>): boolean {
    const [a, b, c, d] = oddNodes;
    
    // Try all three possible pairings:
    // 1. (a-b, c-d)
    if (!hasEdge(a, b, edgeSet) && !hasEdge(c, d, edgeSet)) {
        return true;
    }
    
    // 2. (a-c, b-d)
    if (!hasEdge(a, c, edgeSet) && !hasEdge(b, d, edgeSet)) {
        return true;
    }
    
    // 3. (a-d, b-c)
    if (!hasEdge(a, d, edgeSet) && !hasEdge(b, c, edgeSet)) {
        return true;
    }
    
    return false;
};
```

## Solution 2: With Detailed Comments

```typescript
function isPossible(n: number, edges: number[][]): boolean {
    // Build degree array and edge set for O(1) edge existence check
    const degree = new Array(n + 1).fill(0);
    const edgeSet = new Set<string>();
    
    for (const [a, b] of edges) {
        degree[a]++;
        degree[b]++;
        // Normalize edge representation: smaller node first
        const key = `${Math.min(a, b)},${Math.max(a, b)}`;
        edgeSet.add(key);
    }
    
    // Collect all nodes with odd degree
    const oddDegreeNodes: number[] = [];
    for (let node = 1; node <= n; node++) {
        if (degree[node] % 2 === 1) {
            oddDegreeNodes.push(node);
        }
    }
    
    const oddCount = oddDegreeNodes.length;
    
    // By handshaking lemma, number of odd-degree nodes must be even
    // With at most 2 edges, we can fix at most 4 odd-degree nodes
    
    if (oddCount === 0) {
        // All nodes already have even degree
        return true;
    }
    
    if (oddCount === 2) {
        // Need to add 1 edge to make both nodes even
        const [u, v] = oddDegreeNodes;
        
        // Option 1: Direct edge u-v
        if (!edgeExists(u, v, edgeSet)) {
            return true;
        }
        
        // Option 2: Use intermediate node w
        // Add edges u-w and v-w
        // This makes u and v even, and w's parity flips twice (stays same)
        // So w must have even degree initially
        for (let w = 1; w <= n; w++) {
            if (w !== u && w !== v) {
                if (!edgeExists(u, w, edgeSet) && !edgeExists(v, w, edgeSet)) {
                    return true;
                }
            }
        }
        
        return false;
    }
    
    if (oddCount === 4) {
        // Need to add 2 edges to pair up 4 odd-degree nodes
        // Try all 3 possible pairings
        const [a, b, c, d] = oddDegreeNodes;
        
        // Pairing 1: (a,b) and (c,d)
        if (!edgeExists(a, b, edgeSet) && !edgeExists(c, d, edgeSet)) {
            return true;
        }
        
        // Pairing 2: (a,c) and (b,d)
        if (!edgeExists(a, c, edgeSet) && !edgeExists(b, d, edgeSet)) {
            return true;
        }
        
        // Pairing 3: (a,d) and (b,c)
        if (!edgeExists(a, d, edgeSet) && !edgeExists(b, c, edgeSet)) {
            return true;
        }
        
        return false;
    }
    
    // Any other count is impossible
    // - Odd count violates handshaking lemma
    // - Count > 4 requires more than 2 edges
    return false;
}

function edgeExists(a: number, b: number, edgeSet: Set<string>): boolean {
    const key = `${Math.min(a, b)},${Math.max(a, b)}`;
    return edgeSet.has(key);
};
```

# Key Insights

- **Handshaking lemma:** Sum of degrees = 2 × edges (always even)
- **Odd-degree count:** Must be even
- **Maximum fixable:** At most 4 odd nodes with 2 edges
- **Edge checking:** Use set for O(1) duplicate detection
- **Pairing strategies:** 
  - 2 nodes: direct or via intermediate
  - 4 nodes: 3 possible pairings

# Common Mistakes to Avoid

1. **Not checking for duplicate edges:**
   ```typescript
   // Wrong: might create duplicate
   if (oddCount === 2) {
       return true; // Always?
   }
   
   // Correct: check if edge exists
   if (!hasEdge(a, b, edgeSet)) {
       return true;
   }
   ```

2. **Wrong edge representation:**
   ```typescript
   // Wrong: directional
   edgeSet.add(`${a},${b}`);
   
   // Correct: normalized (undirected)
   edgeSet.add(`${Math.min(a, b)},${Math.max(a, b)}`);
   ```

3. **Forgetting intermediate node option:**
   ```typescript
   // Wrong: only checking direct connection
   if (oddCount === 2) {
       return !hasEdge(a, b, edgeSet);
   }
   
   // Correct: also try via intermediate
   for (let x = 1; x <= n; x++) {
       if (!hasEdge(a, x) && !hasEdge(b, x)) return true;
   }
   ```

4. **Not trying all pairings for 4 nodes:**
   ```typescript
   // Wrong: only one pairing
   return !hasEdge(a, b) && !hasEdge(c, d);
   
   // Correct: try all 3 pairings
   return (a-b, c-d) || (a-c, b-d) || (a-d, b-c);
   ```

# Mathematical Background

**Handshaking Lemma:**
```
Sum of all degrees = 2 × number of edges

Since 2 × edges is even, sum of degrees is even.
Therefore, number of odd-degree nodes must be even.
```

**Adding edges effect:**
```
Adding edge (u, v):
- Increases degree[u] by 1
- Increases degree[v] by 1
- Changes parity of exactly 2 nodes
```
