# Intuition

A **complete connected component** is a subgraph where:
1. All vertices are connected (it's a connected component)
2. Every pair of vertices has an edge (it's complete/clique)

**Key insight:** For a complete component with k vertices, it must have exactly **k × (k-1) / 2** edges (complete graph formula).

---

# Approach

1. **Find all connected components** using DFS/BFS or Union-Find
2. **For each component:**
   - Count vertices (k)
   - Count edges within the component
   - Check if edges = k × (k-1) / 2 (complete graph)
3. **Count** how many components are complete

---

# Complexity

- **Time complexity:** $O(n + e)$ where e is number of edges
- **Space complexity:** $O(n + e)$ for graph and visited tracking

# Code

## Solution 1: DFS with Component Analysis (Recommended)

```typescript
function countCompleteComponents(n: number, edges: number[][]): number {
    // Build adjacency list
    const graph = new Map<number, number[]>();
    for (let i = 0; i < n; i++) {
        graph.set(i, []);
    }
    
    for (const [a, b] of edges) {
        graph.get(a)!.push(b);
        graph.get(b)!.push(a);
    }
    
    const visited = new Array(n).fill(false);
    let completeCount = 0;
    
    // Find each connected component
    for (let i = 0; i < n; i++) {
        if (!visited[i]) {
            const component: number[] = [];
            
            // DFS to find all nodes in this component
            function dfs(node: number) {
                visited[node] = true;
                component.push(node);
                
                for (const neighbor of graph.get(node)!) {
                    if (!visited[neighbor]) {
                        dfs(neighbor);
                    }
                }
            }
            
            dfs(i);
            
            // Check if this component is complete
            const k = component.length;
            const expectedEdges = k * (k - 1) / 2;
            
            // Count actual edges in this component
            let actualEdges = 0;
            for (const node of component) {
                actualEdges += graph.get(node)!.length;
            }
            actualEdges /= 2; // Each edge counted twice
            
            if (actualEdges === expectedEdges) {
                completeCount++;
            }
        }
    }
    
    return completeCount;
};

```

## Solution 2: BFS Approach

```typescript
function countCompleteComponents(n: number, edges: number[][]): number {
    // Build graph
    const graph = new Map<number, Set<number>>();
    for (let i = 0; i < n; i++) {
        graph.set(i, new Set());
    }
    
    for (const [a, b] of edges) {
        graph.get(a)!.add(b);
        graph.get(b)!.add(a);
    }
    
    const visited = new Array(n).fill(false);
    let completeCount = 0;
    
    for (let i = 0; i < n; i++) {
        if (!visited[i]) {
            // BFS to find component
            const component: number[] = [];
            const queue = [i];
            visited[i] = true;
            
            while (queue.length > 0) {
                const node = queue.shift()!;
                component.push(node);
                
                for (const neighbor of graph.get(node)!) {
                    if (!visited[neighbor]) {
                        visited[neighbor] = true;
                        queue.push(neighbor);
                    }
                }
            }
            
            // Check if complete
            const k = component.length;
            const expectedEdges = k * (k - 1) / 2;
            
            let actualEdges = 0;
            for (const node of component) {
                actualEdges += graph.get(node)!.size;
            }
            actualEdges /= 2;
            
            if (actualEdges === expectedEdges) {
                completeCount++;
            }
        }
    }
    
    return completeCount;
};

```

## Solution 3: Union-Find with Component Tracking

```typescript
function countCompleteComponents(n: number, edges: number[][]): number {
    const parent = Array.from({ length: n }, (_, i) => i);
    
    function find(x: number): number {
        if (parent[x] !== x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }
    
    function union(x: number, y: number): void {
        const rootX = find(x);
        const rootY = find(y);
        if (rootX !== rootY) {
            parent[rootX] = rootY;
        }
    }
    
    // Union all edges
    for (const [a, b] of edges) {
        union(a, b);
    }
    
    // Group nodes by component
    const components = new Map<number, number[]>();
    for (let i = 0; i < n; i++) {
        const root = find(i);
        if (!components.has(root)) {
            components.set(root, []);
        }
        components.get(root)!.push(i);
    }
    
    // Count edges per component
    const edgeCount = new Map<number, number>();
    for (const [a, b] of edges) {
        const root = find(a);
        edgeCount.set(root, (edgeCount.get(root) || 0) + 1);
    }
    
    // Check each component
    let completeCount = 0;
    for (const [root, nodes] of components) {
        const k = nodes.length;
        const expectedEdges = k * (k - 1) / 2;
        const actualEdges = edgeCount.get(root) || 0;
        
        if (actualEdges === expectedEdges) {
            completeCount++;
        }
    }
    
    return completeCount;
};

```

## Solution 4: With Detailed Comments

```typescript
function countCompleteComponents(n: number, edges: number[][]): number {
    // Step 1: Build adjacency list representation
    const graph = new Map<number, number[]>();
    for (let i = 0; i < n; i++) {
        graph.set(i, []);
    }
    
    for (const [a, b] of edges) {
        graph.get(a)!.push(b);
        graph.get(b)!.push(a);
    }
    
    const visited = new Array(n).fill(false);
    let completeComponentCount = 0;
    
    // Step 2: Find each connected component
    for (let startNode = 0; startNode < n; startNode++) {
        if (!visited[startNode]) {
            const componentNodes: number[] = [];
            
            // DFS to collect all nodes in this component
            function dfs(node: number) {
                visited[node] = true;
                componentNodes.push(node);
                
                for (const neighbor of graph.get(node)!) {
                    if (!visited[neighbor]) {
                        dfs(neighbor);
                    }
                }
            }
            
            dfs(startNode);
            
            // Step 3: Check if this component is complete
            const vertexCount = componentNodes.length;
            
            // A complete graph with k vertices has k*(k-1)/2 edges
            const requiredEdges = vertexCount * (vertexCount - 1) / 2;
            
            // Count actual edges in this component
            // Each edge is counted twice (once from each endpoint)
            let edgeCount = 0;
            for (const node of componentNodes) {
                edgeCount += graph.get(node)!.length;
            }
            edgeCount /= 2; // Divide by 2 since each edge counted twice
            
            // Step 4: Check if complete
            if (edgeCount === requiredEdges) {
                completeComponentCount++;
            }
        }
    }
    
    return completeComponentCount;
};

```

## Solution 5: Optimized with Degree Check

```typescript
function countCompleteComponents(n: number, edges: number[][]): number {
    const graph = new Map<number, Set<number>>();
    for (let i = 0; i < n; i++) {
        graph.set(i, new Set());
    }
    
    for (const [a, b] of edges) {
        graph.get(a)!.add(b);
        graph.get(b)!.add(a);
    }
    
    const visited = new Array(n).fill(false);
    let count = 0;
    
    for (let i = 0; i < n; i++) {
        if (!visited[i]) {
            const component: number[] = [];
            
            const dfs = (node: number) => {
                visited[node] = true;
                component.push(node);
                for (const neighbor of graph.get(node)!) {
                    if (!visited[neighbor]) {
                        dfs(neighbor);
                    }
                }
            };
            
            dfs(i);
            
            // In a complete graph, every node has degree k-1
            const k = component.length;
            const isComplete = component.every(node => 
                graph.get(node)!.size === k - 1
            );
            
            if (isComplete) {
                count++;
            }
        }
    }
    
    return count;
};

```

# Key Insights

- **Complete graph formula:** k vertices need k×(k-1)/2 edges
- **Degree check:** In complete graph, every node has degree k-1
- **Edge counting:** Each edge counted twice in adjacency list
- **Isolated nodes:** Single nodes are complete (0 edges needed)
- **Connected components:** Find them first, then check completeness

---

# Mathematical Background

**Complete graph (Kₙ):**
```
Vertices: n
Edges: n × (n-1) / 2

Examples:
K₁: 0 edges (single node)
K₂: 1 edge (pair)
K₃: 3 edges (triangle)
K₄: 6 edges (complete quad)
K₅: 10 edges
```

**Why k×(k-1)/2?**
```
Each of k vertices connects to k-1 others
Total connections: k × (k-1)
But each edge counted twice: divide by 2
Result: k × (k-1) / 2
```

# Common Mistakes to Avoid

1. **Not dividing edge count by 2:**
   ```typescript
   // Wrong: counting each edge twice
   let edges = 0;
   for (const node of component) {
       edges += graph.get(node)!.length;
   }
   if (edges === expected) // Wrong!
   
   // Correct: divide by 2
   edges /= 2;
   if (edges === expected) // Correct
   ```

2. **Wrong complete graph formula:**
   ```typescript
   // Wrong: k × k
   const expected = k * k;
   
   // Correct: k × (k-1) / 2
   const expected = k * (k - 1) / 2;
   ```

3. **Not checking all components:**
   ```typescript
   // Wrong: stopping after first component
   if (!visited[0]) {
       // find component and check
       return isComplete ? 1 : 0;
   }
   
   // Correct: check all components
   for (let i = 0; i < n; i++) {
       if (!visited[i]) {
           // find and check each component
       }
   }
   ```

4. **Forgetting isolated nodes:**
   ```typescript
   // Wrong: only checking nodes with edges
   for (const [a, b] of edges) {
       // process only connected nodes
   }
   
   // Correct: check all nodes 0 to n-1
   for (let i = 0; i < n; i++) {
       if (!visited[i]) {
           // isolated nodes are complete too
       }
   }
   ```

| Operation | Time | Notes |
|-----------|------|-------|
| Build graph | $O(e)$ | Process all edges |
| Find components | $O(n + e)$ | DFS/BFS |

| **Total** | **$O(n + e)$** | **Linear** |

| Approach | Pros | Cons |
|----------|------|------|

| BFS | Iterative, clear | More code |
| Union-Find | Elegant grouping | More complex |

# Recommended Solution

**Use Solution 1 (DFS with Edge Counting)** for:
- $O(n + e)$ time complexity
- $O(n + e)$ space complexity
- Clear and intuitive
- Easy to understand
- Interview-friendly

> This is the most straightforward approach for counting complete connected components!
